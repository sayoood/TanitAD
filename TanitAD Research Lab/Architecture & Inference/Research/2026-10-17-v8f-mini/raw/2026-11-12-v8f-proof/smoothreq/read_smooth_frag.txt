def cname(pair, setting, kind, word):
    return "%s-%s-%s-%s-%s" % (PRE, pair, setting, kind, word)


def pair_names(setting, pair, wd):
    out = {}
    for kind in ("C", "U", "F"):
        if wd[kind] is not None:
            out["%s_%s_%s" % (pair, setting, kind)] = cname(pair, setting, kind, wd[kind])
    for f in FAMS:
        out["%s_%s_%s" % (pair, setting, f)] = cname(pair, setting, f, wd["fam"][f])
    return out


def decide(conds):
    """three-valued: False when one condition is False (it decides without the others), None when one is not available, else True"""
    if any(c is False for c in conds):
        return False
    if any(c is None for c in conds):
        return None
    return True


def c_lower(w):
    return None if w is None or w["C"] is None else bool(w["C"] == "LOWER")


def c_higher(w):
    return None if w is None or w["C"] is None else bool(w["C"] == "HIGHER")


def c_nh(w, k):
    return None if w is None or w[k] is None else bool(w[k] in NOT_HIGHER)


def lvl_reaches(rem, shares):
    """rem: list of REMOVES-CONTACT values (True/False/None) of the arms of one smooth level; shares: their pooled shares of the RR-to-RRc contact gap. -> True when every arm removes contact and the mean of the shares is at
    least 0.5 (a share of None counts as not reaching half); False when one arm does not remove contact or the half is not reached; None when a REMOVES value is not available and none is False."""
    d = decide(rem)
    if d is not True:
        return d
    if any(s is None for s in shares):
        return False
    return bool(sum(shares) / len(shares) >= HALF)


def verdict_names(words, shares):
    """words[pair] = {"C","U","F","fam"} or None (withheld); shares = {arm: pooled share of the RR-to-RRc contact gap or None}. -> (names, why)."""
    names, why = {}, {}

    def put(key, cond, yes, no, label):
        if cond is None:
            why[key] = "not written: a word this name needs is not available (withheld by a control or the headway instrument is not qualified): %s" % label
        else:
            names[key] = PRE + "-" + (yes if cond else no)
    rem = {}
    for x in RR_ARMS:
        w = words.get(lab(x) + "-VS-RR")
        rem[x] = decide([c_lower(w), c_nh(w, "U")])
        put("ARM_%s_REMOVES" % lab(x), rem[x], lab(x) + "-REMOVES-CONTACT", lab(x) + "-NOT-SHOWN", lab(x) + " REMOVES-CONTACT")
    found, unresolved = None, None
    for nn, arms in ((50, ("S50",)), (70, ("S70", "S7B")), (90, ("S90",))):
        r = lvl_reaches([rem[x] for x in arms], [shares.get(x) for x in arms])
        if r is None:
            unresolved = nn
            break
        if r is True:
            found = nn
            break
    if found is not None:
        names["VERDICT_SMOOTH_REQUIREMENT"] = "%s-SMOOTH-REQUIREMENT-R%02d" % (PRE, found)
    elif unresolved is not None:
        why["VERDICT_SMOOTH_REQUIREMENT"] = "not written: the REMOVES-CONTACT name of a level-%d arm is not available, and no lower level reached half" % unresolved
    else:
        names["VERDICT_SMOOTH_REQUIREMENT"] = PRE + "-NO-TESTED-SMOOTH-LEVEL-REACHES-HALF"
    put("VERDICT_TIME_COURSE", decide([c_higher(words.get("S7B-VS-W7B"))]), "TIME-COURSE-CONFIRMED", "TIME-COURSE-NOT-CONFIRMED", "TIME-COURSE")
    put("VERDICT_SMOOTH_SPREAD_ALONE", decide([c_lower(words.get("Z00-VS-RR"))]), "SMOOTH-SPREAD-ALONE-REMOVES-CONTACT", "SMOOTH-SPREAD-ALONE-NOT-SHOWN", "SMOOTH-SPREAD-ALONE")
    sh = words.get("TOK-VS-SHUFM")
    put("VERDICT_TK_GAIN", decide([c_lower(sh), c_nh(sh, "U")]), "TK-GAIN-FROM-THE-SCENE", "TK-GAIN-NOT-SHOWN-FROM-THE-SCENE", "TK-GAIN-FROM-THE-SCENE")
    if "VERDICT_TK_GAIN" in names and names["VERDICT_TK_GAIN"].endswith("-NOT-SHOWN-FROM-THE-SCENE"):
        sr = words.get("SHUFM-VS-RR")
        if sr is None or sr["C"] is None:
            why["VERDICT_TK_GAIN_BESIDE_SHUFM_VS_RR"] = "the name of SHUFM-VS-RR (C) is not available (withheld or the headway instrument is not qualified)"
        else:
            names["VERDICT_TK_GAIN_BESIDE_SHUFM_VS_RR"] = cname("SHUFM-VS-RR", "AGE", "C", sr["C"])
    put("VERDICT_TKD_BIAS", decide([c_lower(words.get("TKD-VS-TOK"))]), "TKD-BIAS-COSTS-CONTACT", "TKD-BIAS-COST-NOT-SHOWN", "TKD-BIAS-COSTS-CONTACT")
    return names, why


def failed_sets(controls):
    """sorts the controls that did not pass: glob (inherited: no name, no value), b0, b1, b2, b4 (the card controls) and b3 {setting: failed arms}"""
    f = {"glob": [], "b0": [], "b1": [], "b2": [], "b4": [], "b3": {}}
    for k, v in controls.items():
        if v.get("PASS") is True:
            continue
        if k in ("b0_tables", "b1_tables", "b2_tables"):
            f[k[:2]].append(k)
        elif k.startswith("b4_"):
            f["b4"].append(k)
        elif k.startswith("b3_"):
            bad = sorted({n.split(":")[0] for n, r in (v.get("runs") or {}).items() if r.get("ok") is not True})
            f["b3"][k[3:]] = bad or list(NEW_ARMS)
        else:
            f["glob"].append(k)
    return f


def blocked_reason(failed, st, pair, forms):
    used = {f[len("RRe_"):] for f in forms if f.startswith("RRe_")}
    if failed["b0"] and used & set(B0_ARMS):
        return "withheld: control %s did not pass (guards the planted arms S50, S90, S7B, W7B)" % ", ".join(failed["b0"])
    if failed["b1"] and "Z00" in used:
        return "withheld: control %s did not pass (guards Z00)" % ", ".join(failed["b1"])
    if (failed["b2"] or failed["b4"]) and pair in D_PAIRS:
        return "withheld: control %s did not pass (guards the pairs of the matched surrogate)" % ", ".join(failed["b2"] + failed["b4"])
    if failed["b2"] and "TKD" in used:
        return "withheld: control %s did not pass (guards TKD)" % ", ".join(failed["b2"])
    if used & set(failed["b3"].get(st) or []):
        return "withheld: control b3 did not pass for a run of this pair at %s" % st
    return None


def withheld_set(failed, pair_forms, st="AGE"):
    return {nm for nm, fm in pair_forms.items() if blocked_reason(failed, st, nm, fm)}


def all_names(words, qualified, pair_forms, failed, shares, st="AGE"):
    """-> (names, why). A pair whose guarding control failed has no name and its words are not used by a verdict."""
    names, why, w2 = {}, {}, {}
    if not qualified:
        why["HEADWAY_" + st] = "U, C and F of the setting %s are not reported: the headway instrument is not qualified on its grid" % st
    for pair, w in words.items():
        r = blocked_reason(failed, st, pair, pair_forms[pair])
        if r:
            why["%s_%s" % (pair, st)] = r
            w2[pair] = None
            continue
        w2[pair] = w
        names.update(pair_names(st, pair, w))
    vn, vw = verdict_names(w2, shares)
    names.update(vn)
    why.update(vw)
    return names, why

def ctl_k4(levels, runs, tbj, reason, tol=TOL_K4):
    """levels of the banked runs recomputed here equal the result file of card ADD84 (settings/AGE/levels/<run>/{U,C,F}). The file has to be the result of that card with its controls listed; it is admitted when every control
    but a4_u3_AGE passed (the one that did not is printed); a file with another failed control, or without controls, is not admitted."""
    try:
        if not tbj or tbj.get("card") != TB_CARD:
            return {"PASS": False, "why": reason or "the file is not the result of card ADD84"}
        ctls = tbj.get("controls")
        if not isinstance(ctls, dict) or not ctls:
            return {"PASS": False, "why": "the result file of card ADD84 lists no controls"}
        notpass = sorted(k for k, v in ctls.items() if not (isinstance(v, dict) and v.get("PASS") is True))
        others = [k for k in notpass if k != "a4_u3_AGE"]
        lv = ((tbj.get("settings") or {}).get("AGE") or {}).get("levels")
        if not isinstance(lv, dict):
            return {"PASS": False, "why": "the result file of card ADD84 holds no levels for AGE", "controls_not_passed": notpass}
        d, n = [], 0
        for f in BANK84F + EEST + BANKED:
            if f not in runs:
                continue
            r = runs[f]
            if r not in lv:
                return {"PASS": False, "why": "run %s is not in the levels of the result file of card ADD84" % r, "controls_not_passed": notpass}
            for c in ("U", "C", "F"):
                d += [abs(float(x) - float(y)) for x, y in zip(levels[r][c], lv[r][c])]
            n += 1
        good = bool(d and max(d) <= tol and not others and n >= len(BANK84F))
        return {"PASS": good, "largest_difference": max(d) if d else None, "runs_compared": n, "tolerance": tol, "controls_not_passed": notpass, "controls_not_passed_besides_a4_u3_AGE": others}
    except Exception as ex:                                                                   # noqa: BLE001
        return {"PASS": False, "why": "%s: %s" % (type(ex).__name__, str(ex)[:200])}


def read_tb_json(path, jl):
    if not path:
        return {"ok": False, "reason": "no --tabclose-json was given", "json": None}
    j = jl(path)
    if j is None:
        return {"ok": False, "reason": "the file %s is not there or not readable" % path, "json": None}
    return {"ok": True, "reason": "", "json": j}

def ctl_b4(stats, st):
    """stats: {(form, seed): (c_src present, est_ok present, n looks, n ok) or None}. Over the looks with a lead read present (both logs, both seeds): sd of the STKM input within 0.80 to 1.25 times that of the TK input of the
    banked runs, the mean of the STKM input within 0.15 m/s of the mean of the TK input, share of the STKM looks marked full >= 0.90."""
    have = all(stats.get((f, s)) is not None for f in ("RRe_STKM", "RRe_TK") for s in SEEDS)
    if not have:
        return {"PASS": False, "complete": False, "setting": st}
    cs = np.concatenate([stats[("RRe_STKM", s)][0] for s in SEEDS])
    ct = np.concatenate([stats[("RRe_TK", s)][0] for s in SEEDS])
    okp = np.concatenate([stats[("RRe_STKM", s)][1] for s in SEEDS])
    sd_s, sd_t = (float(np.std(cs)) if len(cs) else None), (float(np.std(ct)) if len(ct) else None)
    ratio = sd_s / sd_t if sd_s is not None and sd_t not in (None, 0.0) else None
    mean_s, mean_t = (float(np.mean(cs)) if len(cs) else None), (float(np.mean(ct)) if len(ct) else None)
    share = float(np.mean(okp)) if len(okp) else None
    c1 = bool(ratio is not None and SD_LO <= ratio <= SD_HI)
    c2 = bool(mean_s is not None and mean_t is not None and abs(mean_s - mean_t) <= MEAN_MAX)
    c3 = bool(share is not None and share >= SHARE_MIN)
    return {"PASS": bool(c1 and c2 and c3), "complete": True, "sd_stkm": sd_s, "sd_tk": sd_t, "sd_ratio": ratio, "mean_stkm": mean_s, "mean_tk": mean_t, "share_full_present_looks": share,
            "sd_ratio_within_0.80_1.25": c1, "mean_within_0.15_of_the_mean_of_TK": c2, "share_at_least_0.90": c3, "setting": st}

# ============================================================================================ tables
def arm_table(levels, runs):
    """per form (the order of the card): C, U, F per seed and pooled (mean over the seeds), and the share of the contact gap between RR and RRc that the form removes, (C_RR - C_X) / (C_RR - C_RRc), per seed and
    pooled (None when the gap is 0). Names nothing."""
    rows = []
    rr, rc = levels[runs["RR"]]["C"], levels[runs["RRc"]]["C"]

    def per(v):
        return [float(x) for x in np.asarray(v, np.float64).reshape(-1)[:len(SEEDS)]]

    def share(c_x, c_rr, c_rc):
        gap = c_rr - c_rc
        return None if gap == 0.0 else (c_rr - c_x) / gap
    for f in TABLE_ORDER:
        if f not in runs:
            continue
        lv = levels[runs[f]]
        row = {"form": f, "run": runs[f]}
        for k in ("C", "U", "F"):
            p = per(lv[k])
            row[k] = {"per_seed": p, "pooled": float(np.mean(p))}
        crr, crc, cx = per(rr), per(rc), row["C"]["per_seed"]
        row["share_of_gap_RR_to_RRc"] = {"per_seed": [share(cx[i], crr[i], crc[i]) for i in range(len(cx))], "pooled": share(row["C"]["pooled"], float(np.mean(crr)), float(np.mean(crc)))}
        rows.append(row)
    return rows


def shares_of(table):
    return {r["form"][len("RRe_"):]: r["share_of_gap_RR_to_RRc"]["pooled"] for r in table if r["form"].startswith("RRe_")}


def split_withheld(part, wset):
    """-> (part without the withheld pairs, withheld_pairs {pair: values}, withheld_table_rows). A table row of an arm of the card whose every pair is withheld (and that has a pair) is moved to the withheld rows, with its
    levels; the levels of that run leave settings/AGE/levels. Nothing of a withheld pair stays in the part."""
    clean = dict(part)
    for key in ("pairs", "cells", "strategic", "words"):
        clean[key] = {n: v for n, v in (part.get(key) or {}).items() if n not in wset}
    wp = {n: {"forms": list(part["pair_forms"][n]), "pair": part["pairs"][n], "cells": part["cells"][n], "strategic": part["strategic"][n]} for n in sorted(wset) if n in part["pairs"]}
    wrows = []
    if isinstance(part.get("table"), list):
        keep, full_forms = [], set()
        arms = set(NEW_FORMS + BANK84F)
        for f in arms:
            fp = [n for n, fm in part["pair_forms"].items() if f in fm]
            if fp and all(n in wset for n in fp):
                full_forms.add(f)
        for r in part["table"]:
            (wrows if r["form"] in full_forms else keep).append(r)
        clean["table"] = keep
        if isinstance(part.get("levels"), dict):
            gone = {r["run"] for r in wrows}
            clean["levels"] = {r: v for r, v in part["levels"].items() if r not in gone}
    return clean, wp, wrows


def render_table(res):
    L = []
    if res.get("names") is None:
        return L
    for st, p in (res.get("settings") or {}).items():
        if not p or not isinstance(p.get("table"), list):
            continue
        L.append("== table by arm %s (levels of the windows with a lead; per seed and pooled; share of the RR-to-RRc contact gap; reported, names nothing)" % st)
        for r in p["table"]:
            sg = r["share_of_gap_RR_to_RRc"]
            fm = lambda v: "n/a" if v is None else "%.3f" % v                                    # noqa: E731
            L.append("   %-9s C %s pooled %.4f  U %s pooled %.4f  F %s pooled %.4f  share %s pooled %s" % (r["form"], ["%.4f" % x for x in r["C"]["per_seed"]], r["C"]["pooled"], ["%.4f" % x for x in r["U"]["per_seed"]],
                                                                                                   r["U"]["pooled"], ["%.4f" % x for x in r["F"]["per_seed"]], r["F"]["pooled"], [fm(x) for x in sg["per_seed"]], fm(sg["pooled"])))
    return L

def run(a):
    global K4
    log = lambda s: print(s, flush=True)                                                       # noqa: E731
    configure()
    g30 = R.jl(os.path.join(a.grid_dir, "grid_p30.json"))
    if g30 is None:
        print("STOP: grid_p30.json in %s is missing" % a.grid_dir)
        return 3
    E = R.engine(a)
    H = E[0]
    J = H.load_join()
    names_ = [str(c) for c in J["clip"]]
    keys = {"AGE": R.grid_keys(g30, names_)}
    bank = R.jl(a.midlook_json)
    e_ok, e_reason, ej = read_e_json(a.estclose_json, R.jl)
    K4 = read_tb_json(a.tabclose_json, R.jl)
    tables, shas, want = {}, {}, []
    for x in NEW_ARMS:
        for s in SEEDS:
            fn = "tab_%s_s%d.npz" % (x, s)
            want.append(fn)
            p = os.path.join(a.tab_dir, fn)
            if os.path.isfile(p):
                tables[(x, s)] = dict(np.load(p, allow_pickle=False))
                shas[fn] = sha_file(p)
            else:
                tables[(x, s)] = None
    rec = R.jl(a.tab_record or os.path.join(a.tab_dir, "SMOOTH_TABLES_RECORD.json"))
    controls = {}
    for c in ("b0", "b1", "b2"):
