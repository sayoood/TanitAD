  333      def evaluate_coll(self, trajs, gt_trajs, segmentation):
  334          '''
  335          trajs: torch.Tensor (B, n_future, 2)
  336          gt_trajs: torch.Tensor (B, n_future, 2)
  337          segmentation: torch.Tensor (B, n_future, 200, 200)
  338          '''

  342

  344          obj_box_coll_sum = torch.zeros(n_future, device=segmentation.device)
  345  
  346          for i in range(B):
  347              gt_box_coll = self.evaluate_single_coll(gt_trajs[i], segmentation[i])
  348  
  349              xx, yy = trajs[i,:,0], trajs[i, :, 1]
  350              yi = ((yy - self.bx[0]) / self.dx[0]).long()
  351              xi = ((xx - self.bx[1]) / self.dx[1]).long()
  352  
  353              m1 = torch.logical_and(
  354                  torch.logical_and(yi >= 0, yi < self.bev_dimension[0]),
  355                  torch.logical_and(xi >= 0, xi < self.bev_dimension[1]),
  356              )
  357              m1 = torch.logical_and(m1, torch.logical_not(gt_box_coll))
  358  
  359              ti = torch.arange(n_future)
  360              obj_coll_sum[ti[m1]] += segmentation[i, ti[m1], yi[m1], xi[m1]].long()
  361  
  362              m2 = torch.logical_not(gt_box_coll)
  363              box_coll = self.evaluate_single_coll(trajs[i], segmentation[i])
  364              obj_box_coll_sum[ti[m2]] += (box_coll[ti[m2]]).long()
  365  
  366          return obj_coll_sum, obj_box_coll_sum
