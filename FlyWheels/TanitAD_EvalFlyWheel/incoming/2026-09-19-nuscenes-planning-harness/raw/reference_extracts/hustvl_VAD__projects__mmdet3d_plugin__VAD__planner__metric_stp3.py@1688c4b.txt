  242      def evaluate_coll(
  243              self, 
  244              trajs, 
  245              gt_trajs, 
  246              segmentation
  247          ):
  248          '''
  249          trajs: torch.Tensor (B, n_future, 2)
  250              自车lidar系为轨迹参考系
  251              ^ y
  252              |
  253              | 
  254              0------->
  255                      x
  256          gt_trajs: torch.Tensor (B, n_future, 2)
  257          segmentation: torch.Tensor (B, n_future, 200, 200)
  258  
  259          '''
  260          B, n_future, _ = trajs.shape
  261          # trajs = trajs * torch.tensor([-1, 1], device=trajs.device)
  262          # gt_trajs = gt_trajs * torch.tensor([-1, 1], device=gt_trajs.device)
  263  
  264          obj_coll_sum = torch.zeros(n_future, device=segmentation.device)
  265          obj_box_coll_sum = torch.zeros(n_future, device=segmentation.device)
  266  
  267          for i in range(B):
  268              gt_box_coll = self.evaluate_single_coll(gt_trajs[i], segmentation[i], input_gt=True)
  269  
  270              xx, yy = trajs[i,:,0], trajs[i, :, 1]
  271              # lidar系下的轨迹转换到图片坐标系下
  272              xi = ((-self.bx[0]/2 - yy) / self.dx[0]).long()
  273              yi = ((-self.bx[1]/2 + xx) / self.dx[1]).long()
  274  
  275              m1 = torch.logical_and(
  276                  torch.logical_and(xi >= 0, xi < self.bev_dimension[0]),
  277                  torch.logical_and(yi >= 0, yi < self.bev_dimension[1]),
  278              ).to(gt_box_coll.device)
  279              m1 = torch.logical_and(m1, torch.logical_not(gt_box_coll))
  280  
  281              ti = torch.arange(n_future)
  282              obj_coll_sum[ti[m1]] += segmentation[i, ti[m1], xi[m1], yi[m1]].long()
  283  
  284              m2 = torch.logical_not(gt_box_coll)
  285              box_coll = self.evaluate_single_coll(trajs[i], segmentation[i], input_gt=False).to(ti.device)
  286              obj_box_coll_sum[ti[m2]] += (box_coll[ti[m2]]).long()
  287  
  288          return obj_coll_sum, obj_box_coll_sum

  290      def compute_L2(self, trajs, gt_trajs):
  291          '''
  292          trajs: torch.Tensor (n_future, 2)

  295          # return torch.sqrt(((trajs[:, :, :2] - gt_trajs[:, :, :2]) ** 2).sum(dim=-1))
  296          pred_len = trajs.shape[0]
  297          ade = float(
  298              sum(
  299                  torch.sqrt(
  300                      (trajs[i, 0] - gt_trajs[i, 0]) ** 2
  301                      + (trajs[i, 1] - gt_trajs[i, 1]) ** 2
  302                  )
  303                  for i in range(pred_len)
  304              )

  306          )
  307          
  308          return ade
